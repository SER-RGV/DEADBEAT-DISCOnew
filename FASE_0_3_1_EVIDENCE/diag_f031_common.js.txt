// Instrumentación de SOLO LECTURA inyectada antes de cualquier script del juego.
// No modifica el HTML: se ejecuta en el contexto del navegador de pruebas.
function initScript(){
  window.__qaEvents=[];
  window.__qaBullets=[];
  window.__qaWa=null;
  window.__qaFrames=0;
  const rec=(e,phase)=>{
    try{
      window.__qaEvents.push({t:Date.now(),phase,type:e.type,
        target:(e.target&&(e.target.id||e.target.tagName))||null,
        pointerType:e.pointerType||null,button:e.button,nullable:null,
        x:e.clientX,y:e.clientY,key:e.key||null,
        prevented:!!e.defaultPrevented, cancelable:!!e.cancelable});
      if(window.__qaEvents.length>1200) window.__qaEvents.shift();
    }catch(_){}
  };
  for(const t of ['pointerdown','pointerup','mousedown','mouseup','click','keydown','keyup']){
    window.addEventListener(t,e=>rec(e,'capture'),true);
    window.addEventListener(t,e=>rec(e,'bubble'),false);
  }
  // Observar la entrada real de proyectiles del JUGADOR en el array del mundo (wa).
  // Firma: objeto con weaponSerial + accuracyCohort (creado en wd()).
  const origPush=Array.prototype.push;
  Array.prototype.push=function(){
    for(let i=0;i<arguments.length;i++){
      const v=arguments[i];
      if(v&&typeof v==='object'&&('weaponSerial'in v)&&('accuracyCohort'in v)){
        window.__qaWa=this;
        window.__qaBullets.push({t:Date.now(),x:Math.round(v.x),y:Math.round(v.y),
          vx:Math.round(v.vx),vy:Math.round(v.vy),r:v.r,weaponId:v.weaponId,
          serial:v.weaponSerial,arrLen:this.length});
        if(window.__qaBullets.length>800) window.__qaBullets.shift();
      }
    }
    return origPush.apply(this,arguments);
  };
  const loop=()=>{window.__qaFrames++;requestAnimationFrame(loop)};
  requestAnimationFrame(loop);
}

async function newSession(browser,log,viewport){
  const ctx=await browser.newContext({viewport:viewport||{width:1280,height:900}});
  const page=await ctx.newPage();
  attach(page,log);
  await page.addInitScript(initScript);
  return {ctx,page};
}
async function load(page,log){
  const t0=Date.now();
  await page.goto(URL,{waitUntil:'load',timeout:180000});
  await page.waitForTimeout(1500);
  const ms=Date.now()-t0;
  if(log) log.loadMs=ms;
  return ms;
}
async function bootToHub(page){
  await page.mouse.click(640,450);
  await page.waitForTimeout(600);
  await page.keyboard.press('Enter');
  await page.waitForTimeout(900);
  await page.keyboard.down('Escape');
  await page.waitForTimeout(2800);
  await page.keyboard.up('Escape');
  await page.waitForTimeout(600);
}
const snap=p=>p.evaluate(SNAP);
